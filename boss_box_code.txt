function xorEnc(t,p){var r='';for(var i=0;i<t.length;i++){r+=String.fromCharCode(t.charCodeAt(i)^p.charCodeAt(i%p.length));}return btoa(r);}
function xorDec(e,p){var t=atob(e),r='';for(var i=0;i<t.length;i++){r+=String.fromCharCode(t.charCodeAt(i)^p.charCodeAt(i%p.length));}return r;}
function hashPin(p){var h=0;for(var i=0;i<p.length;i++){h=((h<<5)-h)+p.charCodeAt(i);h|=0;}return h.toString(36);}

async function getApiKey(){
  if(SESSION_KEY)return SESSION_KEY;
  var e=localStorage.getItem(KEY_STORAGE);
  var h=localStorage.getItem(PIN_HASH_KEY);
  if(!e||!h){
    var k=prompt('🔑 Klistra in din Claude API-nyckel:');
    if(!k||!k.trim())return null;
    var p=prompt('🔒 Välj PIN (minst 4 tecken):');
    if(!p||p.length<4){alert('PIN måste vara minst 4 tecken.');return null;}
    var c=prompt('🔒 Bekräfta PIN:');
    if(p!==c){alert('PIN matchar inte.');return null;}
    localStorage.setItem(KEY_STORAGE,xorEnc(k.trim(),p));
    localStorage.setItem(PIN_HASH_KEY,hashPin(p));
    SESSION_KEY=k.trim();
    return SESSION_KEY;
  }
  var p=prompt('🔒 Ange PIN:');
  if(!p)return null;
  if(hashPin(p)!==h){alert('❌ Fel PIN.');return null;}
  SESSION_KEY=xorDec(e,p);
  return SESSION_KEY;
}

function getMemBox(){
  var l=window.KanvasAPI.listBoxes();
  return l.find(function(b){return b.name===MEMORY_BOX;});
}
function ensureMemBox(){
  if(!getMemBox())window.KanvasAPI.addBox('// Delat minne (Einar & Boss)\n{}',MEMORY_BOX);
}

var tools=[
  {name:'list_boxes',description:'Lista rutor',input_schema:{type:'object',properties:{}}},
  {name:'get_box_code',description:'Hämta kod',input_schema:{type:'object',properties:{idOrName:{type:'string'}},required:['idOrName']}},
  {name:'set_box_code',description:'Ändra kod',input_schema:{type:'object',properties:{idOrName:{type:'string'},code:{type:'string'}},required:['idOrName','code']}},
  {name:'add_box',description:'Skapa ruta',input_schema:{type:'object',properties:{code:{type:'string'},name:{type:'string'}},required:['code']}},
  {name:'run_box',description:'Kör ruta',input_schema:{type:'object',properties:{idOrName:{type:'string'}},required:['idOrName']}},
  {name:'delete_box',description:'Ta bort ruta',input_schema:{type:'object',properties:{idOrName:{type:'string'}},required:['idOrName']}},
  {name:'read_memory',description:'Läs minne',input_schema:{type:'object',properties:{}}},
  {name:'write_memory',description:'Skriv minne',input_schema:{type:'object',properties:{content:{type:'string'}},required:['content']}}
];

function runTool(name,inp){
  try{
    if(name==='list_boxes')return{ok:true,boxes:window.KanvasAPI.listBoxes()};
    if(name==='get_box_code')return{ok:true,code:window.KanvasAPI.getBoxCode(inp.idOrName)};
    if(name==='set_box_code')return{ok:window.KanvasAPI.setBoxCode(inp.idOrName,inp.code)};
    if(name==='add_box')return{ok:true,id:window.KanvasAPI.addBox(inp.code,inp.name)};
    if(name==='run_box')return{ok:window.KanvasAPI.runBoxById(inp.idOrName)};
    if(name==='delete_box')return{ok:window.KanvasAPI.deleteBoxById(inp.idOrName)};
    if(name==='read_memory'){var mb=getMemBox();return{ok:true,mem:mb?window.KanvasAPI.getBoxCode(mb.id):null};}
    if(name==='write_memory'){ensureMemBox();var mb=getMemBox();return{ok:window.KanvasAPI.setBoxCode(mb.id,inp.content)};}
    return{ok:false,error:'Okänt'};
  }catch(e){return{ok:false,error:e.message};}
}

async function sendToClaude(){
  var key=await getApiKey();
  if(!key)throw new Error('Ingen nyckel');
  var res=await fetch('https://api.anthropic.com/v1/messages',{
    method:'POST',
    headers:{'Content-Type':'application/json','x-api-key':key,'anthropic-version':'2023-06-01','anthropic-dangerous-direct-browser-access':'true'},
    body:JSON.stringify({model:'claude-sonnet-4-20250514',max_tokens:1500,system:sysPrompt,messages:messages,tools:tools})
  });
  return res.json();
}

async function handleTurn(){
  var guard=0;
  while(guard++<8){
    var data=await sendToClaude();
    if(!data||!data.content){addMsg('system','Fel: kunde inte tolka svaret.');return;}
    var tus=[],txt=[];
    data.content.forEach(function(b){if(b.type==='text')txt.push(b.text);if(b.type==='tool_use')tus.push(b);});
    if(txt.length)addMsg('boss',txt.join('\n'));
    messages.push({role:'assistant',content:data.content});
    saveChat();
    if(!tus.length)return;
    var results=tus.map(function(tu){
      var r=runTool(tu.name,tu.input||{});
      addMsg('system','→ '+tu.name);
      return{type:'tool_result',tool_use_id:tu.id,content:JSON.stringify(r)};
    });
    messages.push({role:'user',content:results});
    saveChat();
  }
  addMsg('system','Avbröt efter för många steg.');
}

function send(){
  var t=input.value.trim();
  if(!t)return;
  input.value='';
  addMsg('user',t);
  messages.push({role:'user',content:t});
  saveChat();
  sendBtn.disabled=true;
  handleTurn().catch(function(e){addMsg('system','Fel: '+e.message);}).then(function(){sendBtn.disabled=false;});
}

function saveChat(){try{localStorage.setItem(STORAGE_KEY,JSON.stringify(messages.slice(-30)));}catch(e){}}
function loadChat(){
  try{
    var s=localStorage.getItem(STORAGE_KEY);
    if(s){
      messages=JSON.parse(s);
      messages.forEach(function(m){
        if(m.role==='user'&&typeof m.content==='string')addMsg('user',m.content);
        if(m.role==='assistant'){
          var t=(m.content||[]).filter(function(b){return b.type==='text';}).map(function(b){return b.text;});
          if(t.length)addMsg('boss',t.join('\n'));
        }
      });
    }
  }catch(e){}
}
